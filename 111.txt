"""
import numpy as np
from scipy import signal

# ---------------------------------------------------------------------------
# 光学常数

def n_si(lambda_um):
    """
    Si 折射率（Sellmeier，Salzberg & Villa 1957，Tatian 1984 拟合，1.36-11 um）。
    n^2 - 1 = 10.6684293 L^2/(L^2-0.301516485^2)
            + 0.0030434748 L^2/(L^2-1.13475115^2)
            + 1.54133408 L^2/(L^2-1104^2),  L in um.
    """
    L = np.asarray(lambda_um, dtype=float)
    n2 = 1.0
    n2 += 10.6684293 * L**2 / (L**2 - 0.301516485**2)
    n2 += 0.0030434748 * L**2 / (L**2 - 1.13475115**2)
    n2 += 1.54133408 * L**2 / (L**2 - 1104.0**2)
    return np.sqrt(n2)


def n_sic(lambda_um):
    """
    4H-SiC 寻常光折射率（Wang et al. 2013, Laser Photonics Rev. 7:831-838；
    有效 0.4047-5 um）。
    n_o^2 = 1 + 0.20075 L^2/(L^2+12.07224)
              + 5.54861 L^2/(L^2-0.02641)
              + 35.65066 L^2/(L^2-1268.24708),  L in um.
    """
    L = np.asarray(lambda_um, dtype=float)
    n2 = 1.0
    n2 += 0.20075 * L**2 / (L**2 + 12.07224)
    n2 += 5.54861 * L**2 / (L**2 - 0.02641)
    n2 += 35.65066 * L**2 / (L**2 - 1268.24708)
    return np.sqrt(np.clip(n2, 1.0, None))


def sigma_to_lambda_um(sigma_cm):
    """波数(cm^-1) -> 波长(um)。"""
    return 1e4 / np.asarray(sigma_cm, dtype=float)


def refractive_index(material, sigma_cm):
    """按材料名返回折射率数组（sigma 单位 cm^-1）。"""
    lam = sigma_to_lambda_um(sigma_cm)
    if material.lower() == "si":
        return n_si(lam)
    elif material.lower() == "sic":
        return n_sic(lam)
    raise ValueError(f"unknown material: {material}")


def snell_angle(n_i, n_t, theta_i_deg):
    """Snell 定律：返回介质 t 内的折射角（度）。"""
    sin_t = n_i * np.sin(np.deg2rad(theta_i_deg)) / n_t
    sin_t = np.clip(sin_t, -1.0, 1.0)
    return np.rad2deg(np.arcsin(sin_t))


def fresnel_intensity_r(n_i, n_t, theta_i_deg, polarization="avg"):
    """
    Fresnel 界面强度反射率 |r|^2。polarization: 's','p','avg'(非偏振平均)。
    小角度下 s/p 差别小，取平均即可。
    """
    ti = np.deg2rad(theta_i_deg)
    st = np.clip(n_i * np.sin(ti) / n_t, -1.0, 1.0)
    tt = np.arcsin(st)
    ci = np.cos(ti)
    ct = np.cos(tt)
    if polarization == "s":
        rs = (n_i * ci - n_t * ct) / (n_i * ci + n_t * ct)
        return rs**2
    elif polarization == "p":
        rp = (n_t * ci - n_i * ct) / (n_t * ci + n_i * ct)
        return rp**2
    else:  # avg
        rs = (n_i * ci - n_t * ct) / (n_i * ci + n_t * ct)
        rp = (n_t * ci - n_i * ct) / (n_t * ci + n_i * ct)
        return 0.5 * (rs**2 + rp**2)


def phase_delta(sigma_cm, d_cm, n1, theta1_deg):
    """两光束/多光束往返相位差 delta = 4*pi*n1*d*cos(theta1)*sigma。"""
    return 4.0 * np.pi * n1 * d_cm * np.cos(np.deg2rad(theta1_deg)) * np.asarray(sigma_cm, dtype=float)


# ---------------------------------------------------------------------------
# 薄膜干涉反射率模型（模型族）
# ---------------------------------------------------------------------------

def reflectance_two_beam(sigma_cm, d_cm, n1, theta0_deg, R12, phi0=0.0,
                         scale=1.0, offset=0.0):
    """
    两光束干涉反射率（一阶近似，含透射因子）：
    R = R01 + (1-R01)^2 R12 + 2(1-R01) sqrt(R01 R12) cos(delta + phi0)
    R01 由空气/外延界面 Fresnel 计算（近法向 avg）。
    scale/offset 用于吸收标定误差（反射率比例/平移），默认 1/0。
    返回 R（无量纲，0-1 量级）。
    """
    n0 = 1.0
    R01 = fresnel_intensity_r(n0, n1, theta0_deg)
    theta1 = snell_angle(n0, n1, theta0_deg)
    delta = phase_delta(sigma_cm, d_cm, n1, theta1)
    R = R01 + (1 - R01)**2 * R12 + 2 * (1 - R01) * np.sqrt(R01 * R12) * np.cos(delta + phi0)
    return scale * R + offset


def reflectance_airy(sigma_cm, d_cm, n1, theta0_deg, R1, R2, phi0=0.0,
                     scale=1.0, offset=0.0):
    """
    多光束干涉（Airy）反射率：
    R = (R1 + R2 + 2 sqrt(R1 R2) cos(delta+phi0))
        / (1 + R1 R2 + 2 sqrt(R1 R2) cos(delta+phi0))
    R1: 空气/外延界面；R2: 外延/衬底界面。
    """
    n0 = 1.0
    theta1 = snell_angle(n0, n1, theta0_deg)
    delta = phase_delta(sigma_cm, d_cm, n1, theta1)
    a = np.sqrt(R1)
    b = np.sqrt(R2)
    cosd = np.cos(delta + phi0)
    R = (R1 + R2 + 2 * a * b * cosd) / (1.0 + R1 * R2 + 2 * a * b * cosd)
    return scale * R + offset

def detrend_reflectance(sigma, R, method="sg", window=81, poly=3):
    """
    去除反射率慢变包络，返回振荡分量 R_osc 与背景 R_bg。
    方法：SG 平滑作为背景（窗口按波数点数给；数据约 0.48 cm^-1 步长）。
    """
    R = np.asarray(R, dtype=float)
    if method == "sg":
        if window % 2 == 0:
            window += 1
        R_bg = signal.savgol_filter(R, window_length=window, polyorder=poly)
    elif method == "gauss":
        R_bg = gaussian_filter1d(R, sigma=window)
    else:
        raise ValueError(method)
    return R - R_bg, R_bg


def envelope_bounds(sigma, R, win=101):
    """滑动极大/极小包络（用于可见度/波段有效性判断）。"""
    R = np.asarray(R, dtype=float)
    n = len(R)
    up = np.full(n, np.nan)
    lo = np.full(n, np.nan)
    half = win // 2
    for i in range(n):
        a = max(0, i - half)
        b = min(n, i + half + 1)
        up[i] = np.nanmax(R[a:b])
        lo[i] = np.nanmin(R[a:b])
    return up, lo

def fft_fringe_frequency(sigma, R_osc, window="hann"):
    """
    FFT 提取干涉主频（cycles per cm^-1）。
    返回 f0, spec 频率轴, 功率谱, 谐波比 (P(2f0)/P(f0))。
    """
    sig = np.asarray(sigma, dtype=float)
    r = np.asarray(R_osc, dtype=float)
    # 均匀重采样到等步长（数据本身近似等步长）
    step = np.median(np.diff(sig))
    n = len(r)
    if window == "hann":
        w = np.hanning(n)
    elif window == "blackman":
        w = np.blackman(n)
    else:
        w = np.ones(n)
    rw = (r - r.mean()) * w
    fft = np.fft.rfft(rw)
    freq = np.fft.rfftfreq(n, d=step)  # cycles per cm^-1
    power = np.abs(fft) ** 2
    # 主峰（排除直流与低频包络残留；搜索下限为 0.5 个周期 / 全带宽）
    lo = np.searchsorted(freq, 1.0 / (sig[-1] - sig[0]))          # >= 1 个周期
    if lo < 1:
        lo = 1
    idx = lo + np.argmax(power[lo:])
    f0 = freq[idx]
    # 谐波功率：在主峰整数倍附近积分
    def band_power(f_target, half=1.5 * step / n * n / (sig[-1] - sig[0])):
        # 半宽取频率分辨率的 1.5 倍
        half = 1.5 * (1.0 / ((sig[-1] - sig[0]) * 1.0)) * 0.5
        m = (freq > f_target - half) & (freq < f_target + half)
        return float(power[m].sum())
    p1 = band_power(f0)
    p2 = band_power(2 * f0)
    p3 = band_power(3 * f0)
    return f0, freq, power, p1, p2, p3


def find_extrema(sigma, R, mode="max"):
    """局部极值（极大或极小）索引，基于五点比较 + 最小间隔去重。"""
    R = np.asarray(R, dtype=float)
    if mode == "max":
        order = 2
        idx = signal.argrelextrema(R, np.greater_equal, order=order)[0]
    else:
        idx = signal.argrelextrema(R, np.less_equal, order=order)[0]
    return idx


def extrema_phase_regression(sigma, R_osc, n1_arr, theta0_deg, mode="max"):
    """
    色散自洽极值相位回归（H1 候选，报告 §7.2 路线 B）：
    极值序 m 与 X_m = 2*n1(sigma_m)*cos(theta1(sigma_m))*sigma_m 线性回归，斜率 = d。
    返回 d(cm), 斜率误差, 拟合点（供可视化）。
    """
    n0 = 1.0
    idx = find_extrema(sigma, R_osc, mode=mode)
    if len(idx) < 3:
        return None, None, None
    sig_m = sigma[idx]
    n1_m = np.asarray(n1_arr)[idx] if np.ndim(n1_arr) else np.full(len(idx), n1_arr)
    theta1 = snell_angle(n0, n1_m, theta0_deg)
    X = 2 * n1_m * np.cos(np.deg2rad(theta1)) * sig_m
    m = np.arange(len(idx), dtype=float)
    A = np.vstack([X, np.ones_like(X)]).T
    coef, res, *_ = np.linalg.lstsq(A, m, rcond=None)
    d = coef[0]
    # 斜率标准误
    n = len(X)
    yhat = A @ coef
    s2 = np.sum((m - yhat) ** 2) / (n - 2)
    cov = s2 * np.linalg.inv(A.T @ A)
    d_err = float(np.sqrt(cov[0, 0]))
    return float(d), d_err, (sig_m, X, m)

def read_attachment(path, expected_rows=7469, drop_first_artifact=True):
    """读取附件 xlsx（第1行表头，第2列反射率 %）。
    返回 (sigma_cm, R_frac)；R_frac 为无量纲（%/100）。
    首点 R=0 按标定截断伪迹剔除（报告 §3.2 协议）。
    """
    import pandas as pd
    df = pd.read_excel(path, sheet_name=0, header=0)
    assert len(df) == expected_rows, f"{path}: 行数异常 {len(df)}"
    sigma = df.iloc[:, 0].to_numpy(dtype=float)
    R = df.iloc[:, 1].to_numpy(dtype=float) / 100.0
    if drop_first_artifact and len(sigma) > 1 and R[0] == 0.0:
        sigma = sigma[1:]
        R = R[1:]
    # 无效值清理
    ok = np.isfinite(sigma) & np.isfinite(R)
    return sigma[ok], R[ok]

    """
    Si 折射率（Sellmeier，Salzberg & Villa 1957，Tatian 1984 拟合，1.36-11 um）。
    n^2 - 1 = 10.6684293 L^2/(L^2-0.301516485^2)
            + 0.0030434748 L^2/(L^2-1.13475115^2)
            + 1.54133408 L^2/(L^2-1104^2),  L in um.
    """
    L = np.asarray(lambda_um, dtype=float)
    n2 = 1.0
    n2 += 10.6684293 * L**2 / (L**2 - 0.301516485**2)
    n2 += 0.0030434748 * L**2 / (L**2 - 1.13475115**2)
    n2 += 1.54133408 * L**2 / (L**2 - 1104.0**2)
    return np.sqrt(n2)


def n_sic(lambda_um):
    """
    4H-SiC 寻常光折射率（Wang et al. 2013, Laser Photonics Rev. 7:831-838；
    有效 0.4047-5 um）。
    n_o^2 = 1 + 0.20075 L^2/(L^2+12.07224)
              + 5.54861 L^2/(L^2-0.02641)
              + 35.65066 L^2/(L^2-1268.24708),  L in um.
    """
    L = np.asarray(lambda_um, dtype=float)
    n2 = 1.0
    n2 += 0.20075 * L**2 / (L**2 + 12.07224)
    n2 += 5.54861 * L**2 / (L**2 - 0.02641)
    n2 += 35.65066 * L**2 / (L**2 - 1268.24708)
    return np.sqrt(np.clip(n2, 1.0, None))


def sigma_to_lambda_um(sigma_cm):
    """波数(cm^-1) -> 波长(um)。"""
    return 1e4 / np.asarray(sigma_cm, dtype=float)


def refractive_index(material, sigma_cm):
    """按材料名返回折射率数组（sigma 单位 cm^-1）。"""
    lam = sigma_to_lambda_um(sigma_cm)
    if material.lower() == "si":
        return n_si(lam)
    elif material.lower() == "sic":
        return n_sic(lam)
    raise ValueError(f"unknown material: {material}")


def snell_angle(n_i, n_t, theta_i_deg):
    """Snell 定律：返回介质 t 内的折射角（度）。"""
    sin_t = n_i * np.sin(np.deg2rad(theta_i_deg)) / n_t
    sin_t = np.clip(sin_t, -1.0, 1.0)
    return np.rad2deg(np.arcsin(sin_t))


def fresnel_intensity_r(n_i, n_t, theta_i_deg, polarization="avg"):
    """
    Fresnel 界面强度反射率 |r|^2。polarization: 's','p','avg'(非偏振平均)。
    小角度下 s/p 差别小，取平均即可。
    """
    ti = np.deg2rad(theta_i_deg)
    st = np.clip(n_i * np.sin(ti) / n_t, -1.0, 1.0)
    tt = np.arcsin(st)
    ci = np.cos(ti)
    ct = np.cos(tt)
    if polarization == "s":
        rs = (n_i * ci - n_t * ct) / (n_i * ci + n_t * ct)
        return rs**2
    elif polarization == "p":
        rp = (n_t * ci - n_i * ct) / (n_t * ci + n_i * ct)
        return rp**2
    else:  # avg
        rs = (n_i * ci - n_t * ct) / (n_i * ci + n_t * ct)
        rp = (n_t * ci - n_i * ct) / (n_t * ci + n_i * ct)
        return 0.5 * (rs**2 + rp**2)


def phase_delta(sigma_cm, d_cm, n1, theta1_deg):
    """两光束/多光束往返相位差 delta = 4*pi*n1*d*cos(theta1)*sigma。"""
    return 4.0 * np.pi * n1 * d_cm * np.cos(np.deg2rad(theta1_deg)) * np.asarray(sigma_cm, dtype=float)

# 薄膜干涉反射率模型（模型族）
# ---------------------------------------------------------------------------

def reflectance_two_beam(sigma_cm, d_cm, n1, theta0_deg, R12, phi0=0.0,
                         scale=1.0, offset=0.0):
    """
    两光束干涉反射率（一阶近似，含透射因子）：
    R = R01 + (1-R01)^2 R12 + 2(1-R01) sqrt(R01 R12) cos(delta + phi0)
    R01 由空气/外延界面 Fresnel 计算（近法向 avg）。
    scale/offset 用于吸收标定误差（反射率比例/平移），默认 1/0。
    返回 R（无量纲，0-1 量级）。
    """
    n0 = 1.0
    R01 = fresnel_intensity_r(n0, n1, theta0_deg)
    theta1 = snell_angle(n0, n1, theta0_deg)
    delta = phase_delta(sigma_cm, d_cm, n1, theta1)
    R = R01 + (1 - R01)**2 * R12 + 2 * (1 - R01) * np.sqrt(R01 * R12) * np.cos(delta + phi0)
    return scale * R + offset


def reflectance_airy(sigma_cm, d_cm, n1, theta0_deg, R1, R2, phi0=0.0,
                     scale=1.0, offset=0.0):
    """
    多光束干涉（Airy）反射率：
    R = (R1 + R2 + 2 sqrt(R1 R2) cos(delta+phi0))
        / (1 + R1 R2 + 2 sqrt(R1 R2) cos(delta+phi0))
    R1: 空气/外延界面；R2: 外延/衬底界面。
    """
    n0 = 1.0
    theta1 = snell_angle(n0, n1, theta0_deg)
    delta = phase_delta(sigma_cm, d_cm, n1, theta1)
    a = np.sqrt(R1)
    b = np.sqrt(R2)
    cosd = np.cos(delta + phi0)
    R = (R1 + R2 + 2 * a * b * cosd) / (1.0 + R1 * R2 + 2 * a * b * cosd)
    return scale * R + offset

def detrend_reflectance(sigma, R, method="sg", window=81, poly=3):
    """
    去除反射率慢变包络，返回振荡分量 R_osc 与背景 R_bg。
    方法：SG 平滑作为背景（窗口按波数点数给；数据约 0.48 cm^-1 步长）。
    """
    R = np.asarray(R, dtype=float)
    if method == "sg":
        if window % 2 == 0:
            window += 1
        R_bg = signal.savgol_filter(R, window_length=window, polyorder=poly)
    elif method == "gauss":
        R_bg = gaussian_filter1d(R, sigma=window)
    else:
        raise ValueError(method)
    return R - R_bg, R_bg


def envelope_bounds(sigma, R, win=101):
    """滑动极大/极小包络（用于可见度/波段有效性判断）。"""
    R = np.asarray(R, dtype=float)
    n = len(R)
    up = np.full(n, np.nan)
    lo = np.full(n, np.nan)
    half = win // 2
    for i in range(n):
        a = max(0, i - half)
        b = min(n, i + half + 1)
        up[i] = np.nanmax(R[a:b])
        lo[i] = np.nanmin(R[a:b])
    return up, lo

def fft_fringe_frequency(sigma, R_osc, window="hann"):
    """
    FFT 提取干涉主频（cycles per cm^-1）。
    返回 f0, spec 频率轴, 功率谱, 谐波比 (P(2f0)/P(f0))。
    """
    sig = np.asarray(sigma, dtype=float)
    r = np.asarray(R_osc, dtype=float)
    # 均匀重采样到等步长（数据本身近似等步长）
    step = np.median(np.diff(sig))
    n = len(r)
    if window == "hann":
        w = np.hanning(n)
    elif window == "blackman":
        w = np.blackman(n)
    else:
        w = np.ones(n)
    rw = (r - r.mean()) * w
    fft = np.fft.rfft(rw)
    freq = np.fft.rfftfreq(n, d=step)  # cycles per cm^-1
    power = np.abs(fft) ** 2
    # 主峰（排除直流与低频包络残留；搜索下限为 0.5 个周期 / 全带宽）
    lo = np.searchsorted(freq, 1.0 / (sig[-1] - sig[0]))          # >= 1 个周期
    if lo < 1:
        lo = 1
    idx = lo + np.argmax(power[lo:])
    f0 = freq[idx]
    # 谐波功率：在主峰整数倍附近积分
    def band_power(f_target, half=1.5 * step / n * n / (sig[-1] - sig[0])):
        # 半宽取频率分辨率的 1.5 倍
        half = 1.5 * (1.0 / ((sig[-1] - sig[0]) * 1.0)) * 0.5
        m = (freq > f_target - half) & (freq < f_target + half)
        return float(power[m].sum())
    p1 = band_power(f0)
    p2 = band_power(2 * f0)
    p3 = band_power(3 * f0)
    return f0, freq, power, p1, p2, p3


def find_extrema(sigma, R, mode="max"):
    """局部极值（极大或极小）索引，基于五点比较 + 最小间隔去重。"""
    R = np.asarray(R, dtype=float)
    if mode == "max":
        order = 2
        idx = signal.argrelextrema(R, np.greater_equal, order=order)[0]
    else:
        idx = signal.argrelextrema(R, np.less_equal, order=order)[0]
    return idx


def extrema_phase_regression(sigma, R_osc, n1_arr, theta0_deg, mode="max"):
    """
    色散自洽极值相位回归（H1 候选，报告 §7.2 路线 B）：
    极值序 m 与 X_m = 2*n1(sigma_m)*cos(theta1(sigma_m))*sigma_m 线性回归，斜率 = d。
    返回 d(cm), 斜率误差, 拟合点（供可视化）。
    """
    n0 = 1.0
    idx = find_extrema(sigma, R_osc, mode=mode)
    if len(idx) < 3:
        return None, None, None
    sig_m = sigma[idx]
    n1_m = np.asarray(n1_arr)[idx] if np.ndim(n1_arr) else np.full(len(idx), n1_arr)
    theta1 = snell_angle(n0, n1_m, theta0_deg)
    X = 2 * n1_m * np.cos(np.deg2rad(theta1)) * sig_m
    m = np.arange(len(idx), dtype=float)
    A = np.vstack([X, np.ones_like(X)]).T
    coef, res, *_ = np.linalg.lstsq(A, m, rcond=None)
    d = coef[0]
    # 斜率标准误
    n = len(X)
    yhat = A @ coef
    s2 = np.sum((m - yhat) ** 2) / (n - 2)
    cov = s2 * np.linalg.inv(A.T @ A)
    d_err = float(np.sqrt(cov[0, 0]))
    return float(d), d_err, (sig_m, X, m)

def read_attachment(path, expected_rows=7469, drop_first_artifact=True):
    """读取附件 xlsx（第1行表头，第2列反射率 %）。
    返回 (sigma_cm, R_frac)；R_frac 为无量纲（%/100）。
    首点 R=0 按标定截断伪迹剔除（报告 §3.2 协议）。
    """
    import pandas as pd
    df = pd.read_excel(path, sheet_name=0, header=0)
    assert len(df) == expected_rows, f"{path}: 行数异常 {len(df)}"
    sigma = df.iloc[:, 0].to_numpy(dtype=float)
    R = df.iloc[:, 1].to_numpy(dtype=float) / 100.0
    if drop_first_artifact and len(sigma) > 1 and R[0] == 0.0:
        sigma = sigma[1:]
        R = R[1:]
    # 无效值清理
    ok = np.isfinite(sigma) & np.isfinite(R)
    return sigma[ok], R[ok]

    """截取分析窗口。"""
    m = (sigma >= lo_cm) & (sigma <= hi_cm)
    return sigma[m], R[m]

    R01 由空气/外延界面 Fresnel 计算（近法向 avg）。
    scale/offset 用于吸收标定误差（反射率比例/平移），默认 1/0。
    返回 R（无量纲，0-1 量级）。
    """
    n0 = 1.0
    R01 = fresnel_intensity_r(n0, n1, theta0_deg)
    theta1 = snell_angle(n0, n1, theta0_deg)
    delta = phase_delta(sigma_cm, d_cm, n1, theta1)
    R = R01 + (1 - R01)**2 * R12 + 2 * (1 - R01) * np.sqrt(R01 * R12) * np.cos(delta + phi0)
    return scale * R + offset


def reflectance_airy(sigma_cm, d_cm, n1, theta0_deg, R1, R2, phi0=0.0,
                     scale=1.0, offset=0.0):
    """
    多光束干涉（Airy）反射率：
    R = (R1 + R2 + 2 sqrt(R1 R2) cos(delta+phi0))
        / (1 + R1 R2 + 2 sqrt(R1 R2) cos(delta+phi0))
    R1: 空气/外延界面；R2: 外延/衬底界面。
    """
    n0 = 1.0
    theta1 = snell_angle(n0, n1, theta0_deg)
    delta = phase_delta(sigma_cm, d_cm, n1, theta1)
    a = np.sqrt(R1)
    b = np.sqrt(R2)
    cosd = np.cos(delta + phi0)
    R = (R1 + R2 + 2 * a * b * cosd) / (1.0 + R1 * R2 + 2 * a * b * cosd)
    return scale * R + offset

def detrend_reflectance(sigma, R, method="sg", window=81, poly=3):
    """
    去除反射率慢变包络，返回振荡分量 R_osc 与背景 R_bg。
    方法：SG 平滑作为背景（窗口按波数点数给；数据约 0.48 cm^-1 步长）。
    """
    R = np.asarray(R, dtype=float)
    if method == "sg":
        if window % 2 == 0:
            window += 1
        R_bg = signal.savgol_filter(R, window_length=window, polyorder=poly)
    elif method == "gauss":
        R_bg = gaussian_filter1d(R, sigma=window)
    else:
        raise ValueError(method)
    return R - R_bg, R_bg


def envelope_bounds(sigma, R, win=101):
    """滑动极大/极小包络（用于可见度/波段有效性判断）。"""
    R = np.asarray(R, dtype=float)
    n = len(R)
    up = np.full(n, np.nan)
    lo = np.full(n, np.nan)
    half = win // 2
    for i in range(n):
        a = max(0, i - half)
        b = min(n, i + half + 1)
        up[i] = np.nanmax(R[a:b])
        lo[i] = np.nanmin(R[a:b])
    return up, lo

def fft_fringe_frequency(sigma, R_osc, window="hann"):
    """
    FFT 提取干涉主频（cycles per cm^-1）。
    返回 f0, spec 频率轴, 功率谱, 谐波比 (P(2f0)/P(f0))。
    """
    sig = np.asarray(sigma, dtype=float)
    r = np.asarray(R_osc, dtype=float)
    # 均匀重采样到等步长（数据本身近似等步长）
    step = np.median(np.diff(sig))
    n = len(r)
    if window == "hann":
        w = np.hanning(n)
    elif window == "blackman":
        w = np.blackman(n)
    else:
        w = np.ones(n)
    rw = (r - r.mean()) * w
    fft = np.fft.rfft(rw)
    freq = np.fft.rfftfreq(n, d=step)  # cycles per cm^-1
    power = np.abs(fft) ** 2
    # 主峰（排除直流与低频包络残留；搜索下限为 0.5 个周期 / 全带宽）
    lo = np.searchsorted(freq, 1.0 / (sig[-1] - sig[0]))          # >= 1 个周期
    if lo < 1:
        lo = 1
    idx = lo + np.argmax(power[lo:])
    f0 = freq[idx]
    # 谐波功率：在主峰整数倍附近积分
    def band_power(f_target, half=1.5 * step / n * n / (sig[-1] - sig[0])):
        # 半宽取频率分辨率的 1.5 倍
        half = 1.5 * (1.0 / ((sig[-1] - sig[0]) * 1.0)) * 0.5
        m = (freq > f_target - half) & (freq < f_target + half)
        return float(power[m].sum())
    p1 = band_power(f0)
    p2 = band_power(2 * f0)
    p3 = band_power(3 * f0)
    return f0, freq, power, p1, p2, p3


def find_extrema(sigma, R, mode="max"):
    """局部极值（极大或极小）索引，基于五点比较 + 最小间隔去重。"""
    R = np.asarray(R, dtype=float)
    if mode == "max":
        order = 2
        idx = signal.argrelextrema(R, np.greater_equal, order=order)[0]
    else:
        idx = signal.argrelextrema(R, np.less_equal, order=order)[0]
    return idx


def extrema_phase_regression(sigma, R_osc, n1_arr, theta0_deg, mode="max"):
    """
    色散自洽极值相位回归（H1 候选，报告 §7.2 路线 B）：
    极值序 m 与 X_m = 2*n1(sigma_m)*cos(theta1(sigma_m))*sigma_m 线性回归，斜率 = d。
    返回 d(cm), 斜率误差, 拟合点（供可视化）。
    """
    n0 = 1.0
    idx = find_extrema(sigma, R_osc, mode=mode)
    if len(idx) < 3:
        return None, None, None
    sig_m = sigma[idx]
    n1_m = np.asarray(n1_arr)[idx] if np.ndim(n1_arr) else np.full(len(idx), n1_arr)
    theta1 = snell_angle(n0, n1_m, theta0_deg)
    X = 2 * n1_m * np.cos(np.deg2rad(theta1)) * sig_m
    m = np.arange(len(idx), dtype=float)
    A = np.vstack([X, np.ones_like(X)]).T
    coef, res, *_ = np.linalg.lstsq(A, m, rcond=None)
    d = coef[0]
    # 斜率标准误
    n = len(X)
    yhat = A @ coef
    s2 = np.sum((m - yhat) ** 2) / (n - 2)
    cov = s2 * np.linalg.inv(A.T @ A)
    d_err = float(np.sqrt(cov[0, 0]))
    return float(d), d_err, (sig_m, X, m)

def g_parameter(R1, R2):
    """多光束显著度参数 G = 4√(R1R2)/(1-R1R2)。"""
    R12 = R1 * R2
    return 4.0 * np.sqrt(R12) / (1.0 - R12)


def near_zero_analysis(sig, R, material, theta0, s_lo, s_hi, order=12):
    """低波数区近零极小值分析：定位最小 R，反推两光束 R12（闭合解）、
    Airy R2（极小值匹配）、G；并检验两光束极大值是否违反能量守恒。"""
    m = (sig >= s_lo) & (sig <= s_hi)
    seg_s, seg_r = sig[m], R[m]
    imin = argrelextrema(seg_r, np.less_equal, order=order)[0]
    if len(imin) == 0:
        return None
    i = imin[np.argmin(seg_r[imin])]
    s_min = float(seg_s[i])
    R_min = float(seg_r[i])
    n1 = refractive_index(material, s_min)
    R01 = float(fresnel_intensity_r(1.0, n1, theta0))
    # 两光束闭合解：R_min = (√R01 - (1-R01)√R12)^2
    sqR12 = (np.sqrt(R01) - np.sqrt(max(R_min, 0.0))) / (1 - R01)
    R12_2b = float(max(sqR12, 0.0) ** 2)
    Max_2b = R01 + (1 - R01) ** 2 * R12_2b + 2 * (1 - R01) * np.sqrt(R01 * R12_2b)
    # Airy：R2 使极小值匹配（(√R1-√R2)^2/(1-√R1√R2)^2 = R_min）
    # 数值解
    def airy_min(R2):
        r1, r2 = np.sqrt(R01), np.sqrt(R2)
        return (r1 - r2) ** 2 / (1 - r1 * r2) ** 2
    from scipy.optimize import brentq
    if airy_min(0.999) > R_min:
        R2_air = float(brentq(lambda r: airy_min(r) - R_min, R_min, 0.999))
    else:

    G_air = g_parameter(R01, R2_air) if R2_air else None
    Max_air = None
    if R2_air:
        r1, r2 = np.sqrt(R01), np.sqrt(R2_air)
        Max_air = float((R01 + R2_air + 2 * r1 * r2) / (1 + R01 * R2_air + 2 * r1 * r2))
    return {"sigma_min": s_min, "R_min": R_min, "R01": R01,

def g_profile(sig, R, material, theta0, s_lo, s_hi, step=250):
    """逐段 G(σ)：局部包络幅值→R12→G。"""
    out = []
    for a in range(s_lo, s_hi, step):
        b = a + step
        m = (sig >= a) & (sig < b)
        if m.sum() < 200:
            continue
        ss, rr = sig[m], R[m]
        imax = argrelextrema(rr, np.greater_equal, order=10)[0]
        imin = argrelextrema(rr, np.less_equal, order=10)[0]
        if len(imax) < 2 or len(imin) < 2:
            continue
        A = 0.5 * (np.median(rr[imax]) - np.median(rr[imin]))
        n1 = refractive_index(material, ss).mean()
        R01 = fresnel_intensity_r(1.0, n1, theta0)
        R12 = (A / (2 * (1 - R01) * np.sqrt(R01))) ** 2 if A > 0 else 0.0
        G = g_parameter(R01, R12) if R12 > 0 else 0.0
        out.append({"sigma_mid": 0.5 * (a + b), "amp_frac": float(A),
                    "R12": float(R12), "G": float(G)})
    return out

    Rfit = B @ coef
    resid_std = float(np.sqrt(np.mean((Rg - Rfit) ** 2)))
    sigma_boot = max(sigma_noise, resid_std / np.sqrt(2.0))
    boot = []
    for _ in range(n):
        Rb = Rfit + rng.normal(0, sigma_boot, size=len(Rg))
        d, _ = d_fft_xspace_hi(Xg, Rb)
        boot.append(d * 1e4)
    return {"n": n, "sigma_noise_abs": sigma_noise, "resid_std_abs": resid_std,
            "sigma_boot_abs": sigma_boot,
            "std_um": float(np.std(boot)),
            "ci95_um": [float(np.percentile(boot, 2.5)),
                        float(np.percentile(boot, 97.5))]}


def fit_airy_cross(sig, R, material, theta0, win, d_guess):
    """Airy 形状拟合交叉校验（中窗口，自由 R2/phi0/scale/offset）。"""
    sig_w, R_w = select_window(sig, R, win[0], win[1])
    n1 = refractive_index(material, sig_w)
    R01 = fresnel_intensity_r(1.0, n1, theta0)

    def resid(p):
        d, z, phi0, scale, offset = p
        R2 = 0.5 / (1 + np.exp(-z))
        Rm = reflectance_airy(sig_w, d, n1, theta0, R01, R2, phi0, scale, offset)
        return Rm - R_w

    p0 = [d_guess, np.log(0.25 / 0.25), 0.0, 1.0, 0.0]
    res = least_squares(resid, p0, method="lm", max_nfev=60000, xtol=1e-13, ftol=1e-13)
    R2 = 0.5 / (1 + np.exp(-res.x[1]))
    return {"d_um": res.x[0] * 1e4, "R2": float(R2), "phi0": float(res.x[2]),
            "scale": float(res.x[3]), "offset": float(res.x[4]),
            "rmse_frac": float(np.sqrt(np.mean(res.fun ** 2)))}

"""
问题3 多光束干涉分析：硅晶圆（附件3/4）+ 碳化硅复核（附件1/2）
================================================================
多光束必要条件（报告 §7.3）：
  ① 界面反射率乘积 R1·R2 足够大：显著度参数 G = 4√(R1R2)/(1-R1R2) 不满足 G<<1。
     G 度量 Airy 级数中一阶之后各项的相对贡献；G<<1 时两光束模型成立。
  ② 层内吸收弱：α d << 1（外延层透明）。
  ③ 光源相干长度 Lc ≥ 2 n1 d cosθ1。
  ④ 两界面平行度：楔角使光程差展宽远小于条纹周期。

检测结论（结构性证据，见下）：
  Si（附件3/4）在低波数区出现多光束干涉：
    · σ≈941 cm^-1 处 R 降至 0.13%，R_min/R01 ≈ 0.004，接近完全相消；

def g_parameter(R1, R2):
    """多光束显著度参数 G = 4√(R1R2)/(1-R1R2)。"""
    R12 = R1 * R2
    return 4.0 * np.sqrt(R12) / (1.0 - R12)


def near_zero_analysis(sig, R, material, theta0, s_lo, s_hi, order=12):
    """低波数区近零极小值分析：定位最小 R，反推两光束 R12（闭合解）、
    Airy R2（极小值匹配）、G；并检验两光束极大值是否违反能量守恒。"""
    m = (sig >= s_lo) & (sig <= s_hi)
    seg_s, seg_r = sig[m], R[m]
    imin = argrelextrema(seg_r, np.less_equal, order=order)[0]
    if len(imin) == 0:
        return None
    i = imin[np.argmin(seg_r[imin])]
    s_min = float(seg_s[i])
    R_min = float(seg_r[i])
    n1 = refractive_index(material, s_min)
    R01 = float(fresnel_intensity_r(1.0, n1, theta0))
    # 两光束闭合解：R_min = (√R01 - (1-R01)√R12)^2
    sqR12 = (np.sqrt(R01) - np.sqrt(max(R_min, 0.0))) / (1 - R01)
    R12_2b = float(max(sqR12, 0.0) ** 2)
    Max_2b = R01 + (1 - R01) ** 2 * R12_2b + 2 * (1 - R01) * np.sqrt(R01 * R12_2b)
    # Airy：R2 使极小值匹配（(√R1-√R2)^2/(1-√R1√R2)^2 = R_min）
    # 数值解
    def airy_min(R2):
        r1, r2 = np.sqrt(R01), np.sqrt(R2)
        return (r1 - r2) ** 2 / (1 - r1 * r2) ** 2
    from scipy.optimize import brentq
    if airy_min(0.999) > R_min:
        R2_air = float(brentq(lambda r: airy_min(r) - R_min, R_min, 0.999))
    else:
        R2_air = None
    G_air = g_parameter(R01, R2_air) if R2_air else None
    Max_air = None
    if R2_air:
        r1, r2 = np.sqrt(R01), np.sqrt(R2_air)
        Max_air = float((R01 + R2_air + 2 * r1 * r2) / (1 + R01 * R2_air + 2 * r1 * r2))
    return {"sigma_min": s_min, "R_min": R_min, "R01": R01,
            "R_min_over_R01": R_min / R01,
            "R12_twobeam_closure": R12_2b,
            "Max_twobeam_closure": Max_2b,
            "energy_conservation_violated_2b": bool(Max_2b > 1.0),
            "R2_airy_min_match": R2_air, "G_airy": G_air,
            "Max_airy": Max_air, "Min_airy": float(airy_min(R2_air)) if R2_air else None}


def g_profile(sig, R, material, theta0, s_lo, s_hi, step=250):
    """逐段 G(σ)：局部包络幅值→R12→G。"""
    out = []
    for a in range(s_lo, s_hi, step):
        b = a + step
        m = (sig >= a) & (sig < b)
        if m.sum() < 200:
            continue
        ss, rr = sig[m], R[m]
        imax = argrelextrema(rr, np.greater_equal, order=10)[0]
        imin = argrelextrema(rr, np.less_equal, order=10)[0]
        if len(imax) < 2 or len(imin) < 2:
            continue
        A = 0.5 * (np.median(rr[imax]) - np.median(rr[imin]))
        n1 = refractive_index(material, ss).mean()
        R01 = fresnel_intensity_r(1.0, n1, theta0)
        R12 = (A / (2 * (1 - R01) * np.sqrt(R01))) ** 2 if A > 0 else 0.0
        G = g_parameter(R01, R12) if R12 > 0 else 0.0
        out.append({"sigma_mid": 0.5 * (a + b), "amp_frac": float(A),
                    "R12": float(R12), "G": float(G)})
    return out


def d_fft_windows(sig, R, material, theta0, windows):
    """多窗口 X 空间 FFT d 估计。"""
    rows = []
    for win in windows:
        sig_w, R_w = select_window(sig, R, win[0], win[1])
        Xg, Rg = xspace_resample(sig_w, R_w, material, theta0)
        d, fwhm = d_fft_xspace_hi(Xg, Rg)
        rows.append({"win": list(win), "d_um": d * 1e4, "fwhm_um": fwhm * 1e4,
                     "cycles": float(d * (Xg[-1] - Xg[0]))})
    return rows


def weighted_d(rows):
    """按周期数加权平均。"""
    w = np.array([max(r["cycles"], 0.5) for r in rows])
    ds = np.array([r["d_um"] for r in rows])
    return float(np.sum(w * ds) / np.sum(w))


def bootstrap_unc(sig, R, material, theta0, win, d0, sigma_noise, n=N_BOOT):
    """自举：全局正弦模型（基频固定）+ 高斯噪声重采样，估计 d 统计不确定。"""
    rng = np.random.default_rng(SEED)
    sig_w, R_w = select_window(sig, R, win[0], win[1])
    Xg, Rg = xspace_resample(sig_w, R_w, material, theta0)
    basis = [np.ones_like(Xg), Xg, Xg ** 2,
             np.cos(2 * np.pi * d0 * Xg), np.sin(2 * np.pi * d0 * Xg)]
    B = np.vstack(basis).T
    coef, *_ = np.linalg.lstsq(B, Rg, rcond=None)
    Rfit = B @ coef
    resid_std = float(np.sqrt(np.mean((Rg - Rfit) ** 2)))
    sigma_boot = max(sigma_noise, resid_std / np.sqrt(2.0))
    boot = []
    for _ in range(n):
        Rb = Rfit + rng.normal(0, sigma_boot, size=len(Rg))
        d, _ = d_fft_xspace_hi(Xg, Rb)
        boot.append(d * 1e4)
    return {"n": n, "sigma_noise_abs": sigma_noise, "resid_std_abs": resid_std,
            "sigma_boot_abs": sigma_boot,
            "std_um": float(np.std(boot)),
            "ci95_um": [float(np.percentile(boot, 2.5)),
                        float(np.percentile(boot, 97.5))]}


def fit_airy_cross(sig, R, material, theta0, win, d_guess):
    """Airy 形状拟合交叉校验（中窗口，自由 R2/phi0/scale/offset）。"""
    sig_w, R_w = select_window(sig, R, win[0], win[1])
    n1 = refractive_index(material, sig_w)
    R01 = fresnel_intensity_r(1.0, n1, theta0)

    def resid(p):
        d, z, phi0, scale, offset = p
        R2 = 0.5 / (1 + np.exp(-z))
        Rm = reflectance_airy(sig_w, d, n1, theta0, R01, R2, phi0, scale, offset)
        return Rm - R_w

    p0 = [d_guess, np.log(0.25 / 0.25), 0.0, 1.0, 0.0]
    res = least_squares(resid, p0, method="lm", max_nfev=60000, xtol=1e-13, ftol=1e-13)
    R2 = 0.5 / (1 + np.exp(-res.x[1]))
    return {"d_um": res.x[0] * 1e4, "R2": float(R2), "phi0": float(res.x[2]),
            "scale": float(res.x[3]), "offset": float(res.x[4]),
            "rmse_frac": float(np.sqrt(np.mean(res.fun ** 2)))}

        for model in ["twobeam", "airy"]:
            def resid(p, model=model):
                d, z, phi0, scale, offset = p
                R2 = 0.5 / (1 + np.exp(-z))
                if model == "airy":
                    Rm = reflectance_airy(sig_w, d, n1, theta0, R01, R2, phi0, scale, offset)
                else:
                    Rm = reflectance_two_beam(sig_w, d, n1, theta0, R2, phi0, scale, offset)
                return Rm - R_w
            res = least_squares(resid, [d0, np.log(0.05 / 0.45), 0.0, 1.0, 0.0],
                                method="lm", max_nfev=40000)

def g_parameter(R1, R2):
    """多光束显著度参数 G = 4√(R1R2)/(1-R1R2)。"""
    R12 = R1 * R2
    return 4.0 * np.sqrt(R12) / (1.0 - R12)


def near_zero_analysis(sig, R, material, theta0, s_lo, s_hi, order=12):
    """低波数区近零极小值分析：定位最小 R，反推两光束 R12（闭合解）、
    Airy R2（极小值匹配）、G；并检验两光束极大值是否违反能量守恒。"""
    m = (sig >= s_lo) & (sig <= s_hi)
    seg_s, seg_r = sig[m], R[m]
    imin = argrelextrema(seg_r, np.less_equal, order=order)[0]
    if len(imin) == 0:
        return None
    i = imin[np.argmin(seg_r[imin])]
    s_min = float(seg_s[i])
    R_min = float(seg_r[i])
    n1 = refractive_index(material, s_min)
    R01 = float(fresnel_intensity_r(1.0, n1, theta0))
    # 两光束闭合解：R_min = (√R01 - (1-R01)√R12)^2
    sqR12 = (np.sqrt(R01) - np.sqrt(max(R_min, 0.0))) / (1 - R01)
    R12_2b = float(max(sqR12, 0.0) ** 2)
    Max_2b = R01 + (1 - R01) ** 2 * R12_2b + 2 * (1 - R01) * np.sqrt(R01 * R12_2b)
    # Airy：R2 使极小值匹配（(√R1-√R2)^2/(1-√R1√R2)^2 = R_min）
    # 数值解
    def airy_min(R2):
        r1, r2 = np.sqrt(R01), np.sqrt(R2)
        return (r1 - r2) ** 2 / (1 - r1 * r2) ** 2
    from scipy.optimize import brentq
    if airy_min(0.999) > R_min:
        R2_air = float(brentq(lambda r: airy_min(r) - R_min, R_min, 0.999))
    else:
        R2_air = None
    G_air = g_parameter(R01, R2_air) if R2_air else None
    Max_air = None
    if R2_air:
        r1, r2 = np.sqrt(R01), np.sqrt(R2_air)
        Max_air = float((R01 + R2_air + 2 * r1 * r2) / (1 + R01 * R2_air + 2 * r1 * r2))
    return {"sigma_min": s_min, "R_min": R_min, "R01": R01,
            "R_min_over_R01": R_min / R01,
            "R12_twobeam_closure": R12_2b,
            "Max_twobeam_closure": Max_2b,
            "energy_conservation_violated_2b": bool(Max_2b > 1.0),
            "R2_airy_min_match": R2_air, "G_airy": G_air,
            "Max_airy": Max_air, "Min_airy": float(airy_min(R2_air)) if R2_air else None}


def g_profile(sig, R, material, theta0, s_lo, s_hi, step=250):
    """逐段 G(σ)：局部包络幅值→R12→G。"""
    out = []
    for a in range(s_lo, s_hi, step):
        b = a + step
        m = (sig >= a) & (sig < b)
        if m.sum() < 200:
            continue
        ss, rr = sig[m], R[m]
        imax = argrelextrema(rr, np.greater_equal, order=10)[0]
        imin = argrelextrema(rr, np.less_equal, order=10)[0]
        if len(imax) < 2 or len(imin) < 2:
            continue
        A = 0.5 * (np.median(rr[imax]) - np.median(rr[imin]))
        n1 = refractive_index(material, ss).mean()
        R01 = fresnel_intensity_r(1.0, n1, theta0)
        R12 = (A / (2 * (1 - R01) * np.sqrt(R01))) ** 2 if A > 0 else 0.0
        G = g_parameter(R01, R12) if R12 > 0 else 0.0
        out.append({"sigma_mid": 0.5 * (a + b), "amp_frac": float(A),
                    "R12": float(R12), "G": float(G)})
    return out


def d_fft_windows(sig, R, material, theta0, windows):
    """多窗口 X 空间 FFT d 估计。"""
    rows = []
    for win in windows:
        sig_w, R_w = select_window(sig, R, win[0], win[1])
        Xg, Rg = xspace_resample(sig_w, R_w, material, theta0)
        d, fwhm = d_fft_xspace_hi(Xg, Rg)
        rows.append({"win": list(win), "d_um": d * 1e4, "fwhm_um": fwhm * 1e4,
                     "cycles": float(d * (Xg[-1] - Xg[0]))})
    return rows


def weighted_d(rows):
    """按周期数加权平均。"""
    w = np.array([max(r["cycles"], 0.5) for r in rows])
    ds = np.array([r["d_um"] for r in rows])
    return float(np.sum(w * ds) / np.sum(w))


def bootstrap_unc(sig, R, material, theta0, win, d0, sigma_noise, n=N_BOOT):
    """自举：全局正弦模型（基频固定）+ 高斯噪声重采样，估计 d 统计不确定。"""
    rng = np.random.default_rng(SEED)
    sig_w, R_w = select_window(sig, R, win[0], win[1])
    Xg, Rg = xspace_resample(sig_w, R_w, material, theta0)
    basis = [np.ones_like(Xg), Xg, Xg ** 2,
             np.cos(2 * np.pi * d0 * Xg), np.sin(2 * np.pi * d0 * Xg)]
    B = np.vstack(basis).T
    coef, *_ = np.linalg.lstsq(B, Rg, rcond=None)
    Rfit = B @ coef
    resid_std = float(np.sqrt(np.mean((Rg - Rfit) ** 2)))
    sigma_boot = max(sigma_noise, resid_std / np.sqrt(2.0))
    boot = []
    for _ in range(n):
        Rb = Rfit + rng.normal(0, sigma_boot, size=len(Rg))
        d, _ = d_fft_xspace_hi(Xg, Rb)
        boot.append(d * 1e4)
    return {"n": n, "sigma_noise_abs": sigma_noise, "resid_std_abs": resid_std,
            "sigma_boot_abs": sigma_boot,
            "std_um": float(np.std(boot)),
            "ci95_um": [float(np.percentile(boot, 2.5)),
                        float(np.percentile(boot, 97.5))]}


def fit_airy_cross(sig, R, material, theta0, win, d_guess):
    """Airy 形状拟合交叉校验（中窗口，自由 R2/phi0/scale/offset）。"""
    sig_w, R_w = select_window(sig, R, win[0], win[1])
    n1 = refractive_index(material, sig_w)
    R01 = fresnel_intensity_r(1.0, n1, theta0)

    def resid(p):
        d, z, phi0, scale, offset = p
        R2 = 0.5 / (1 + np.exp(-z))
        Rm = reflectance_airy(sig_w, d, n1, theta0, R01, R2, phi0, scale, offset)
        return Rm - R_w

    p0 = [d_guess, np.log(0.25 / 0.25), 0.0, 1.0, 0.0]
    res = least_squares(resid, p0, method="lm", max_nfev=60000, xtol=1e-13, ftol=1e-13)
    R2 = 0.5 / (1 + np.exp(-res.x[1]))
    return {"d_um": res.x[0] * 1e4, "R2": float(R2), "phi0": float(res.x[2]),
            "scale": float(res.x[3]), "offset": float(res.x[4]),
            "rmse_frac": float(np.sqrt(np.mean(res.fun ** 2)))}
